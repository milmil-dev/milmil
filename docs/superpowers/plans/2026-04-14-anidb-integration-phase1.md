# AniDB Integration — Phase 1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Teach milmil to recognize AniDB anime IDs, automatically enrich cross-site IDs (Bangumi / AniList / MAL / TMDB / Kitsu) after every match, and add a title-based fallback matcher pass using a locally cached `anime-titles.xml` — laying groundwork for a later Phase 2 UDP + ED2K matcher.

**Architecture:** A new `api/internal/integration/anidb` package owns two in-memory indexes refreshed every 24 hours: a Manami cross-ID mapping and an AniDB titles fuzzy-search index. A scheduler job refreshes caches; the matcher invokes the service after every successful pass for ID enrichment and as a new Pass 4 when the existing passes (download rules → DandanPlay hash → Bangumi title → TMDB) all fail. The frontend surfaces the new `anidb_id` as an external link.

**Tech Stack:** Go 1.24, SQLite + `sqlc`, `golang-migrate`, standard-library XML + JSON decoding (`encoding/xml`, `encoding/json`), `net/http`, existing `cache.Cache` + `ws.Hub`, React 19 + TanStack Query on the frontend.

```go
package anidb

import (
    "encoding/json"
    "regexp"
    "strconv"
)

type Source string

const (
    SourceAniDB   Source = "anidb"
    SourceMAL     Source = "mal"
    SourceAniList Source = "anilist"
    SourceKitsu   Source = "kitsu"
    SourceTMDB    Source = "tmdb"
    SourceBangumi Source = "bangumi"
)

// IDSet is the cross-site identifier set for a single anime.
// Zero means unknown.
type IDSet struct {
    AniDB   int64
    MAL     int64
    AniList int64
    Kitsu   int64
    TMDB    int64
    Bangumi int64
}

// Mapping is an in-memory index built from the Manami offline database.
type Mapping struct {
    bySource map[Source]map[int64]IDSet
}

var sourcePatterns = []struct {
    src Source
    re  *regexp.Regexp
}{
    {SourceAniDB,   regexp.MustCompile(`anidb\.net/anime/(\d+)`)},
    {SourceMAL,     regexp.MustCompile(`myanimelist\.net/anime/(\d+)`)},
    {SourceAniList, regexp.MustCompile(`anilist\.co/anime/(\d+)`)},
    {SourceKitsu,   regexp.MustCompile(`kitsu\.(?:io|app)/anime/(\d+)`)},
    {SourceTMDB,    regexp.MustCompile(`themoviedb\.org/(?:tv|movie)/(\d+)`)},
}

type manamiDoc struct {
    Data []struct {
        Sources []string `json:"sources"`
    } `json:"data"`
}

// LoadMapping parses a Manami anime-offline-database JSON blob.
func LoadMapping(raw []byte) (*Mapping, error) {
    var doc manamiDoc
    if err := json.Unmarshal(raw, &doc); err != nil {
        return nil, err
    }
    m := &Mapping{bySource: make(map[Source]map[int64]IDSet, len(sourcePatterns))}
    for _, sp := range sourcePatterns {
        m.bySource[sp.src] = make(map[int64]IDSet)
    }
    for _, entry := range doc.Data {
        set := IDSet{}
        for _, url := range entry.Sources {
            for _, sp := range sourcePatterns {
                if mm := sp.re.FindStringSubmatch(url); mm != nil {
                    id, err := strconv.ParseInt(mm[1], 10, 64)
                    if err != nil { continue }
                    setField(&set, sp.src, id)
                }
            }
        }
        // Index the entry under every source that has an ID.
        for _, sp := range sourcePatterns {
            if id := getField(&set, sp.src); id != 0 {
                m.bySource[sp.src][id] = set
            }
        }
    }
    return m, nil
}

// Resolve returns the full IDSet for (source, id).
func (m *Mapping) Resolve(src Source, id int64) (IDSet, bool) {
    if m == nil { return IDSet{}, false }
    bucket, ok := m.bySource[src]
    if !ok { return IDSet{}, false }
    set, ok := bucket[id]
    return set, ok
}

func setField(s *IDSet, src Source, v int64) {
    switch src {
    case SourceAniDB:   s.AniDB = v
    case SourceMAL:     s.MAL = v
    case SourceAniList: s.AniList = v
    case SourceKitsu:   s.Kitsu = v
    case SourceTMDB:    s.TMDB = v
    case SourceBangumi: s.Bangumi = v
    }
}

func getField(s *IDSet, src Source) int64 {
    switch src {
    case SourceAniDB:   return s.AniDB
    case SourceMAL:     return s.MAL
    case SourceAniList: return s.AniList
    case SourceKitsu:   return s.Kitsu
    case SourceTMDB:    return s.TMDB
    case SourceBangumi: return s.Bangumi
    }
    return 0
}
```

```go
package anidb

import (
    "encoding/xml"
    "sort"
    "strings"
    "unicode"

    "golang.org/x/text/unicode/norm"
)

type Candidate struct {
    AniDBID int64
    Title   string
    Score   float64
}

type titleEntry struct {
    aid        int64
    normalized string // NFKC + lowercase + alnum-only
    raw        string
}

type TitleIndex struct {
    entries []titleEntry
}

type xmlDoc struct {
    Anime []struct {
        Aid    int64 `xml:"aid,attr"`
        Titles []struct {
            Value string `xml:",chardata"`
        } `xml:"title"`
    } `xml:"anime"`
}

// LoadTitleIndex parses an anime-titles.xml document.
func LoadTitleIndex(raw []byte) (*TitleIndex, error) {
    var doc xmlDoc
    if err := xml.Unmarshal(raw, &doc); err != nil {
        return nil, err
    }
    idx := &TitleIndex{}
    for _, a := range doc.Anime {
        for _, t := range a.Titles {
            normed := normalizeTitle(t.Value)
            if normed == "" { continue }
            idx.entries = append(idx.entries, titleEntry{
                aid:        a.Aid,
                normalized: normed,
                raw:        strings.TrimSpace(t.Value),
            })
        }
    }
    return idx, nil
}

func normalizeTitle(s string) string {
    s = norm.NFKC.String(s)
    s = strings.ToLower(s)
    var b strings.Builder
    for _, r := range s {
        if unicode.IsLetter(r) || unicode.IsDigit(r) {
            b.WriteRune(r)
        }
    }
    return b.String()
}

// Search returns candidates ranked by score desc. year is currently
// a soft hint; when 0 it's ignored.
func (idx *TitleIndex) Search(query string, year int) []Candidate {
    if idx == nil { return nil }
    q := normalizeTitle(query)
    if q == "" { return nil }

    // One candidate per aid; keep highest-scoring title per anime.
    best := make(map[int64]Candidate, 16)
    for _, e := range idx.entries {
        score := similarity(q, e.normalized)
        if score < 0.5 { continue }
        if existing, ok := best[e.aid]; ok && existing.Score >= score { continue }
        best[e.aid] = Candidate{AniDBID: e.aid, Title: e.raw, Score: score}
    }

    out := make([]Candidate, 0, len(best))
    for _, c := range best { out = append(out, c) }
    sort.Slice(out, func(i, j int) bool { return out[i].Score > out[j].Score })
    return out
}

// similarity returns a score in [0,1]. 1.0 means equal after normalization.
// Otherwise uses longest common substring length over max length.
func similarity(a, b string) float64 {
    if a == b { return 1.0 }
    lcs := longestCommonSubstring(a, b)
    if lcs == 0 { return 0 }
    maxLen := len(a)
    if len(b) > maxLen { maxLen = len(b) }
    return float64(lcs) / float64(maxLen)
}

func longestCommonSubstring(a, b string) int {
    ar, br := []rune(a), []rune(b)
    if len(ar) == 0 || len(br) == 0 { return 0 }
    prev := make([]int, len(br)+1)
    curr := make([]int, len(br)+1)
    best := 0
    for i := 1; i <= len(ar); i++ {
        for j := 1; j <= len(br); j++ {
            if ar[i-1] == br[j-1] {
                curr[j] = prev[j-1] + 1
                if curr[j] > best { best = curr[j] }
            } else {
                curr[j] = 0
            }
        }
        prev, curr = curr, prev
    }
    return best
}
```

```go
    "io"
    "net/http"
)

const (
    DefaultManamiURL = "https://raw.githubusercontent.com/manami-project/anime-offline-database/master/anime-offline-database.json"
    DefaultTitlesURL = "https://anidb.net/api/anime-titles.xml.gz"
    userAgent        = "milmil-anidb-phase1/1.0"
    maxReadBytes     = 200 * 1024 * 1024 // 200MB safety cap
)

type Client interface {
    DownloadManami(ctx context.Context) ([]byte, error)
    DownloadTitles(ctx context.Context) ([]byte, error)
}

type httpClient struct {
    http       *http.Client
    manamiURL  string
    titlesURL  string
}

func NewClient(h *http.Client, manamiURL, titlesURL string) Client {
    if h == nil { h = http.DefaultClient }
    if manamiURL == "" { manamiURL = DefaultManamiURL }
    if titlesURL == "" { titlesURL = DefaultTitlesURL }
    return &httpClient{http: h, manamiURL: manamiURL, titlesURL: titlesURL}
}

func (c *httpClient) DownloadManami(ctx context.Context) ([]byte, error) {
    return c.fetch(ctx, c.manamiURL, false)
}

func (c *httpClient) DownloadTitles(ctx context.Context) ([]byte, error) {
    return c.fetch(ctx, c.titlesURL, true)
}

func (c *httpClient) fetch(ctx context.Context, url string, maybeGzip bool) ([]byte, error) {
    req, err := http.NewRequestWithContext(ctx, http.MethodGet, url, nil)
    if err != nil { return nil, err }
    req.Header.Set("User-Agent", userAgent)
    resp, err := c.http.Do(req)
    if err != nil { return nil, err }
    defer resp.Body.Close()
    if resp.StatusCode != http.StatusOK {
        return nil, fmt.Errorf("anidb: GET %s returned %d", url, resp.StatusCode)
    }

    var reader io.Reader = io.LimitReader(resp.Body, maxReadBytes)
    if maybeGzip {
        // anime-titles is gzip-encoded at the body level (not transfer-encoding)
        gr, gzErr := gzip.NewReader(reader)
        if gzErr == nil {
            defer gr.Close()
            reader = gr
        }
        // If gzip.NewReader fails, fall back to raw body (covers tests that serve plain text).
    }
    return io.ReadAll(reader)
}
```

```go
package anidb

import (
    "context"
    "encoding/json"
    "errors"
    "log/slog"
    "os"
    "path/filepath"
    "sync"
    "time"
)

// Service is the facade the rest of the app depends on. Safe for concurrent use.
// When caches are missing (cold start or failed download), Resolve/SearchTitles
// return empty results — they are intentionally non-blocking.
type Service struct {
    client Client
    dir    string

    mu       sync.RWMutex
    mapping  *Mapping
    titles   *TitleIndex

    minInterval  time.Duration
    lastFetchMu  sync.Mutex
    lastFetch    time.Time
}

type lastFetchFile struct {
    Manami time.Time `json:"manami"`
    Titles time.Time `json:"titles"`
}

func NewService(c Client, dataDir string) *Service {
    return &Service{
        client:      c,
        dir:         dataDir,
        minInterval: 24 * time.Hour,
    }
}

func (s *Service) SetMinInterval(d time.Duration) { s.minInterval = d }

func (s *Service) Resolve(src Source, id int64) (IDSet, bool) {
    s.mu.RLock(); defer s.mu.RUnlock()
    return s.mapping.Resolve(src, id)
}

func (s *Service) SearchTitles(query string, year int) []Candidate {
    s.mu.RLock(); defer s.mu.RUnlock()
    return s.titles.Search(query, year)
}

// Refresh downloads both files if the last fetch is older than minInterval,
// writes them atomically to dataDir, and swaps the in-memory indexes.
func (s *Service) Refresh(ctx context.Context) error {
    s.lastFetchMu.Lock()
    if time.Since(s.lastFetch) < s.minInterval && !s.lastFetch.IsZero() {
        s.lastFetchMu.Unlock()
        return nil
    }
    s.lastFetchMu.Unlock()

    if s.client == nil { return errors.New("anidb: no client configured") }
    if err := os.MkdirAll(s.dir, 0o755); err != nil { return err }

    manamiBytes, err := s.client.DownloadManami(ctx)
    if err != nil { return err }
    titlesBytes, err := s.client.DownloadTitles(ctx)
    if err != nil { return err }

    // Parse first — never swap partial state.
    m, err := LoadMapping(manamiBytes)
    if err != nil { return err }
    ti, err := LoadTitleIndex(titlesBytes)
    if err != nil { return err }

    if err := atomicWrite(filepath.Join(s.dir, "manami.json"), manamiBytes); err != nil { return err }
    if err := atomicWrite(filepath.Join(s.dir, "titles.xml"), titlesBytes); err != nil { return err }

    s.mu.Lock()
    s.mapping = m
    s.titles = ti
    s.mu.Unlock()

    now := time.Now().UTC()
    s.lastFetchMu.Lock()
    s.lastFetch = now
    s.lastFetchMu.Unlock()

    lf := lastFetchFile{Manami: now, Titles: now}
    if buf, err := json.Marshal(lf); err == nil {
        _ = atomicWrite(filepath.Join(s.dir, "last-fetch.json"), buf)
    }
    slog.Info("anidb: refresh complete", "manami_bytes", len(manamiBytes), "titles_bytes", len(titlesBytes))
    return nil
}

// LoadFromDisk hydrates indexes from a previous refresh, without any network.
func (s *Service) LoadFromDisk() error {
    manamiBytes, err := os.ReadFile(filepath.Join(s.dir, "manami.json"))
    if err != nil { return err }
    titlesBytes, err := os.ReadFile(filepath.Join(s.dir, "titles.xml"))
    if err != nil { return err }

    m, err := LoadMapping(manamiBytes)
    if err != nil { return err }
    ti, err := LoadTitleIndex(titlesBytes)
    if err != nil { return err }

    s.mu.Lock()
    s.mapping = m
    s.titles = ti
    s.mu.Unlock()

    if buf, err := os.ReadFile(filepath.Join(s.dir, "last-fetch.json")); err == nil {
        var lf lastFetchFile
        if json.Unmarshal(buf, &lf) == nil {
            s.lastFetchMu.Lock()
            s.lastFetch = lf.Manami
            s.lastFetchMu.Unlock()
        }
    }
    return nil
}

func atomicWrite(path string, data []byte) error {
    tmp := path + ".tmp"
    if err := os.WriteFile(tmp, data, 0o644); err != nil { return err }
    return os.Rename(tmp, path)
}
```

```go
// EnrichExternalIDs fills any NULL external-ID columns on an anime row using
// the Manami mapping. Never overwrites an existing value. seed is the IDSet
// known so far (typically from a successful match).
func (m *Matcher) EnrichExternalIDs(ctx context.Context, animeID string, seed anidb.IDSet) error {
    if m.anidb == nil { return nil }

    merged := seed
    // Walk each source; if the seed knows it, ask mapping to expand.
    for _, src := range []anidb.Source{anidb.SourceAniDB, anidb.SourceBangumi, anidb.SourceAniList, anidb.SourceMAL, anidb.SourceTMDB} {
        id := seedField(seed, src)
        if id == 0 { continue }
        if set, ok := m.anidb.Resolve(src, id); ok {
            if merged.AniDB == 0   { merged.AniDB   = set.AniDB }
            if merged.MAL == 0     { merged.MAL     = set.MAL }
            if merged.AniList == 0 { merged.AniList = set.AniList }
            if merged.TMDB == 0    { merged.TMDB    = set.TMDB }
            if merged.Bangumi == 0 { merged.Bangumi = set.Bangumi }
        }
    }

    params := store.UpdateAnimeExternalIDsParams{ID: animeID}
    if merged.AniDB != 0   { params.AnidbID   = sql.NullInt64{Int64: merged.AniDB,   Valid: true} }
    if merged.AniList != 0 { params.AnilistID = sql.NullInt64{Int64: merged.AniList, Valid: true} }
    if merged.Bangumi != 0 { params.BangumiID = sql.NullInt64{Int64: merged.Bangumi, Valid: true} }
    if merged.MAL != 0     { params.MalID     = sql.NullInt64{Int64: merged.MAL,     Valid: true} }
    if merged.TMDB != 0    { params.TmdbID    = sql.NullInt64{Int64: merged.TMDB,    Valid: true} }
    return m.queries.UpdateAnimeExternalIDs(ctx, params)
}

func seedField(s anidb.IDSet, src anidb.Source) int64 {
    switch src {
    case anidb.SourceAniDB:   return s.AniDB
    case anidb.SourceAniList: return s.AniList

    case anidb.SourceMAL:     return s.MAL
    case anidb.SourceTMDB:    return s.TMDB
```

```go
// --- Pass 4: AniDB titles fallback ---
if m.anidb != nil {
    const acceptThreshold = 0.75
    const ambiguityMargin = 0.05

    for _, f := range files {
        if matched[f.ID] { continue }
        parsed := fileparse.Parse(f.Filename)
        if parsed.Title == "" { continue }

        cands := m.anidb.SearchTitles(parsed.Title, parsed.Year)
        if len(cands) == 0 { continue }
        top := cands[0]
        if top.Score < acceptThreshold { continue }
        if len(cands) > 1 {
            gap := top.Score - cands[1].Score
            if gap < ambiguityMargin*top.Score { continue }
        }

        set, ok := m.anidb.Resolve(anidb.SourceAniDB, top.AniDBID)
        if !ok { set = anidb.IDSet{AniDB: top.AniDBID} }

        // Retry Bangumi / TMDB with resolved IDs.
        var animeID string
        var retryErr error
        if set.Bangumi != 0 && m.bangumi != nil {
            animeID, retryErr = m.upsertAnimeByBangumi(ctx, f, set.Bangumi, parsed)
        }
        if animeID == "" && set.TMDB != 0 && m.tmdb != nil {
            animeID, retryErr = m.upsertAnimeByTMDB(ctx, f, set.TMDB, parsed)
        }
        if animeID == "" && retryErr == nil {
            animeID, retryErr = m.upsertAnimeByAnidb(ctx, f, top, parsed)
        }
        if retryErr != nil {
            summary.Errors++
            slog.Warn("matcher: pass4 upsert failed", "file", f.ID, "err", retryErr)
            continue
        }
        if animeID == "" { continue }

        if err := m.EnrichExternalIDs(ctx, animeID, set); err != nil {
            slog.Warn("matcher: enrich failed", "anime", animeID, "err", err)
        }
        summary.Matched++
        summary.ByAnidbTitle++
        matched[f.ID] = true

        processed++
        emit(scanner.ProgressEvent{
            Type: "match:progress", LibraryID: libraryID,
            FilesMatched: summary.Matched, FilesTotal: total,
            CurrentFile: f.Filename,
        })
    }
}
```
